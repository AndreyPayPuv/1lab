#define _CRT_SECURE_NO_WARNINGS
#include <stdio.h>
#include <string.h>

struct student
{
    char famil[20];
    char name[20];
    char facult[20];
    int Nomzach;
};

int main(void)
{
    struct student stud[3] = {
        {"Ivanov", "Petr", "FIT", 12345},
        {"Sidorov", "Alexei", "FEN", 54321},
        {"Kozlova", "Marina", "FIT", 67890}
    };

    char searchFamil[20] = { 0 };
    int i;
    int found = 0;

    printf("Surname: ");
    if (scanf("%19s", searchFamil) != 1)
    {
        printf("Error\n");
        return 1;
    }
    searchFamil[19] = '\0';

    for (i = 0; i < 3; i++)
    {
        if (strcmp(stud[i].famil, searchFamil) == 0)
        {
            printf("Found student: %s %s, faculty: %s, student ID number: %d\n",
                stud[i].famil, stud[i].name, stud[i].facult, stud[i].Nomzach);
            found = 1;
        }
    }

    if (!found)
        printf("The student with the surname %s is not found.\n", searchFamil);

    return 0;
}
