#define _CRT_SECURE_NO_WARNINGS
#include <stdio.h>
#include <stdlib.h>
#include <time.h>
 
int main(void)
{
    int n, i;
    int* a;
 
    printf("Size: ");
    if (scanf("%d", &n) != 1 || n <= 0)
    {
        printf("Error size\n");
        return 1;
    }
 
    a = (int*)malloc(n * sizeof(int));
    if (a == NULL)
    {
        printf("Error memory\n");
        return 1;
    }
 
    srand((unsigned int)time(NULL));
    for (i = 0; i < n; i++)
        a[i] = rand() % 100;
 
    printf("Result:\n", n);
    for (i = 0; i < n; i++)
        printf("%d ", a[i]);
    printf("\n");
 
    free(a);
 
    return 0;
}
